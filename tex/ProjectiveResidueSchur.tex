    The fermionic sub-programme of Cosmochrony locates the three-generation mass split in the
    $J_3$-odd part of the squared projective endomorphism $E_\Pi^2$ restricted to the gauge-singlet
    generation triplet $\Cgen$, parametrised by a single real number $u$ through
    $E_\Pi^2|_{\Cgen}=\mathrm{diag}(1,\tfrac12+u,\tfrac12-u)$, with the even sector
    $\mathrm{diag}(1,\tfrac12,\tfrac12)$, the algebraic value of $(C_2-J_3^2)/C_2$ at $C_2=2$.
    Its reading as the Born--Infeld even sector would require an identification between the
    conditional $3\times3$ model of~\cite{Beau2026a34} on $\operatorname{Sym}^2(V_\rho)$ and
    $(E_\Pi^2)_{\mathrm{even}}$; no such identification is available, so that reading is not used
    here.
    This note fixes the structural status of $u$ before any explicit construction of $E_\Pi$.
    First, the projected Dirac square admits a universal Feshbach/Schur form
    $E_\Pi=-\PiS\,D\,(1-P)\,D\,\PiS^{*}=-M^{\dagger}M$ with $P=\PiS^{*}\PiS$, exhibiting $E_\Pi$ as the
    Schur complement of the spinorial directions eliminated by the non-injective projection, negative
    semi-definite and vanishing in the injective limit.
    Second, the chiral block decomposition shows that $u$ is controlled by the $\Deff^{\pm}$-transported,
    generation-projected part of the antiunitary chiral equivariance defect
    \[
      \Delta_\chi(P)=\pi_{LL}-\tau\,\overline{\pi_{RR}}\,\tau^{-1}
    \]
    of the eliminated block $1-P$, rather than by a naive block difference.
    Third, the minimal non-injectivity $c\leftrightarrow q-c$ is chirally symmetric, so $u=0$ at the
    level of axioms A1--A3; a non-zero $u$ requires a chiral symmetry-breaking carried by the
    projection-locking axiom A4.
    The finite locking sector is proved $J_\Pi$-equivariant, so $u_{\mathrm{fin}}=0$, and chirality is
    shown to be a Lorentzian rather than a finite-fibre datum.
    Finally, the Schur-transversality branch required by the Born--Infeld genus companion is closed in the
    present Lorentzian spin stratum: the projected A4 commutator has the exact metaplectic opening
    \[
      \alpha(t,s)=ts\,\frac{r}{\sinh r}=ts+O\!\left((ts)^2\right),\qquad \mu(t,s)\equiv0,
    \]
    with $\cosh r=1+\tfrac{ts}{2}$, and its timelike Clifford symbol is not in the transported zero-mode
    locus.
    Together with the electric-genus result of the A4 companion, this promotes the existence statement to
    $u\neq0$ in the present stratum.
    The remaining open deliverables are the explicit Lorentzian eliminated block $1-P(s)$, the absolute
    normalisation $|u|$, and the projected Yukawa sector.

  \section{The projective residue as a Schur complement}
    \label{sec:schur}

    Let $\PiS:\mathcal S\to\mathcal S_\Pi$ be the restriction of the admissible projection to the
    spinorial bundle, $\PiS^{*}$ the admissible minimal lift, and
    \begin{equation}
      \label{eq:dpidef}
      D_{\Pi,g,A}:=\PiS\circ D_{g,A}\circ \PiS^{*},\qquad P:=\PiS^{*}\PiS,
    \end{equation}
    following~\cite{Beau2026q14}.
    One does not assume $\PiS\PiS^{*}=\mathrm{id}_{\mathcal S_\Pi}$; this holds only in the injectively
    resolved limit.
    Symbol-compatibility means the projection defect affects only the zero-order part of
    $D^2_{\Pi,g,A}$, leaving the Clifford principal symbol intact.

    \begin{proposition}[Schur form of the projective residue]
      \label{prop:schur}
      Under symbol-compatibility, the projective endomorphism of~\cite{Beau2026q14} satisfies
      \begin{equation}
        \label{eq:schur}
        \EPi=-\PiS\,D_{g,A}\,(1-P)\,D_{g,A}\,\PiS^{*}=-M^{\dagger}M,
        \qquad M:=(1-P)\,D_{g,A}\,\PiS^{*}.
      \end{equation}
      In particular $\EPi\preceq 0$, it is zero-order, and $\EPi=0$ if and only if $1-P=0$, i.e.\ in the
      injective limit.
    \end{proposition}

    \begin{proof}
      With $P=\PiS^{*}\PiS$ idempotent, $D^2_{\Pi,g,A}=\PiS D_{g,A}P D_{g,A}\PiS^{*}$.
      The uncompressed square $\PiS D_{g,A}^2\PiS^{*}$ reconstructs the effective spin--gauge Lichnerowicz
      operator on $\mathcal S_\Pi$ up to symbol, so that
      \begin{equation*}
        \mathrm{Lich}(g,A)\approx \PiS D_{g,A}^2\PiS^{*}=D^2_{\Pi,g,A}+\PiS D_{g,A}(1-P)D_{g,A}\PiS^{*}.
      \end{equation*}
      Subtracting the Lichnerowicz block from $D^2_{\Pi,g,A}$ gives~\eqref{eq:schur}.
      Since $1-P$ is a self-adjoint projector and $D_{g,A}$ is self-adjoint,
      $\PiS D_{g,A}(1-P)D_{g,A}\PiS^{*}=M^{\dagger}M\succeq 0$, hence $\EPi=-M^{\dagger}M\preceq 0$.
      The first-order part vanishes by symbol-compatibility, leaving a zero-order endomorphism, and
      $\EPi=0\Leftrightarrow M=0\Leftrightarrow(1-P)D_{g,A}\PiS^{*}=0$, which for a faithful symbol holds
      exactly when $1-P=0$.
    \end{proof}

    Proposition~\ref{prop:schur} makes precise the statement of~\cite{Beau2026q14} that $E_\Pi$ is not a
    postulated finite Dirac datum but is induced by the non-injective projection itself: it is the Schur
    complement of the eliminated sector $1-P$.

  \section{Chiral block reduction and the localisation of $u$}
    \label{sec:chiral}

    The chirality operator $\gamma_5=i\gamma^0\gamma^1\gamma^2\gamma^3$ of the effective Clifford algebra
    exists as a global section because of the Lorentzian closure of the effective
    metric~\cite{Beau2026q11}.
    The Dirac operator is chirality-odd, hence off-diagonal in the chiral splitting
    $\mathcal S_\Pi=\mathcal S_L\oplus\mathcal S_R$:
    \begin{equation}
      \label{eq:blocks}
      D_{g,A}=\begin{pmatrix}0&\Deff^{-}\\ \Deff^{+}&0\end{pmatrix},
      \qquad
      1-P=\begin{pmatrix}\pi_{LL}&\pi_{LR}\\ \pi_{RL}&\pi_{RR}\end{pmatrix},
    \end{equation}
    with $\pi_{LL},\pi_{RR}$ self-adjoint and $\pi_{RL}=\pi_{LR}^{\dagger}$.

    \begin{proposition}[Chiral diagonal blocks of the residue]
      \label{prop:chiralblocks}
      With the conventions of~\eqref{eq:blocks},
      \begin{equation}
        \label{eq:diag}
        E_{LL}=-\PiS\,\Deff^{-}\pi_{RR}\Deff^{+}\,\PiS^{*},\qquad
        E_{RR}=-\PiS\,\Deff^{+}\pi_{LL}\Deff^{-}\,\PiS^{*}.
      \end{equation}
      The left-admissibility condition $P_R\EPi P_R=0$ of~\cite{Beau2026q14} is equivalent to $\pi_{LL}=0$:
      the non-injective projection eliminates no purely left-handed direction.
    \end{proposition}

    \begin{proof}
      Direct multiplication of~\eqref{eq:blocks} gives
      \begin{equation*}
        D_{g,A}(1-P)D_{g,A}=
        \begin{pmatrix}\Deff^{-}\pi_{RR}\Deff^{+}&\Deff^{-}\pi_{RL}\Deff^{-}\\
          \Deff^{+}\pi_{LR}\Deff^{+}&\Deff^{+}\pi_{LL}\Deff^{-}\end{pmatrix}.
      \end{equation*}
      Conjugating by $\PiS,\PiS^{*}$ and using $\EPi=-\PiS D_{g,A}(1-P)D_{g,A}\PiS^{*}$
      gives~\eqref{eq:diag}.
      The right-right block of $\EPi$ is $E_{RR}=-\PiS\Deff^{+}\pi_{LL}\Deff^{-}\PiS^{*}$, which vanishes
      for a faithful Weyl symbol exactly when $\pi_{LL}=0$.
    \end{proof}

    \begin{corollary}[The left-admissible residue is chiral block-diagonal]
      \label{cor:block-diagonal}
      On the left-admissible branch $\pi_{LL}=0$, the projective residue $\EPi$ is chiral block-diagonal,
      $E_{LR}=E_{RL}=0$, so $[\EPi,\gamma_5]=0$.
    \end{corollary}

    \begin{proof}
      Since $1-P$ is a self-adjoint projector it is positive semidefinite, and a positive-semidefinite
      self-adjoint operator with a vanishing diagonal block has vanishing off-diagonal blocks; hence
      $\pi_{LL}=0$ forces $\pi_{LR}=\pi_{RL}=0$.
      By~\eqref{eq:blocks} the off-diagonal blocks of $\EPi$ are then $-\PiS\Deff^{-}\pi_{RL}\Deff^{-}\PiS^{*}=0$
      and its adjoint, so $\EPi=\mathrm{diag}(E_{LL},0)$ and commutes with $\gamma_5$.
      Equivalently, this follows from $\EPi=-M^{\dagger}M\preceq0$ together with $E_{RR}=0$:
      a negative-semidefinite self-adjoint operator with a vanishing diagonal block has vanishing off-diagonal
      blocks.
      The chiral asymmetry of the residue is therefore carried by the diagonal imbalance
      $E_{LL}$ versus $E_{RR}=0$, not by a chiral commutator.
    \end{proof}

    \begin{corollary}[Localisation of $u$]
      \label{cor:localise}
      After restriction to $\Cgen$, the $J_3$-odd part of $E_\Pi^2$ is controlled by the chiral asymmetry
      of the eliminated block,
      \begin{equation}
        \label{eq:uasym}
        u\ \propto\ \pi_{RR}-\pi_{LL},
      \end{equation}
      dressed by the Weyl transport $\Deff^{\pm}$.
      The split $u$ is therefore an intrinsic spectral asymmetry of the chiral part of the projective
      residue, not a shell average of any front cocycle.
    \end{corollary}

    This raw block expression is refined in Section~\ref{sec:spinorial-locking}, where the antiunitary
    nature of $J_\Pi$ replaces the naive block difference by the transported defect $\Delta_\chi(P)$.

  \section{Stratification: $u$ vanishes unless A4 breaks chiral symmetry}
    \label{sec:strat}

    The candidate minimal non-injectivity is the parity involution $c\leftrightarrow q-c$ on
    conjugate Weil blocks, a structurally motivated hypothesis: O18 establishes Born--Infeld
    parity as a covariance of the response family and states the fibre identification as an open
    problem~\cite{Beau2026fe}.
    Its spinorial lift is the antilinear Born--Infeld parity $J_\Pi$, which exchanges the chiral
    subbundles, $J_\Pi P_L=P_R J_\Pi$~\cite{Beau2026q14}.
    A parity that exchanges $\mathcal S_L\leftrightarrow\mathcal S_R$ treats the two chiralities
    symmetrically, so the minimal eliminated block is chirally symmetric,
    $\pi_{LL}=\pi_{RR}$.

    \begin{proposition}[Chiral-symmetry stratification of $u$]
      \label{prop:strat}
      If $\pi_{LL}=\pi_{RR}$ then $u=0$.
      The conjugate pair $\{V^{(c)},V^{(q-c)}\}$ carries no preferred orientation at the level of axioms
      A1--A3, consistently with the non-premature-selection axiom A3~\cite{Beau2026m}.
      Hence at the A1--A3 level $u=0$.
      A non-zero $u$ requires $\pi_{LL}\neq\pi_{RR}$, a chiral symmetry-breaking that can originate only in
      the projection-locking axiom A4.
    \end{proposition}

    \begin{proof}
      By Corollary~\ref{cor:localise}, $u\propto\pi_{RR}-\pi_{LL}$, so $\pi_{LL}=\pi_{RR}$ forces $u=0$.
      The minimal non-injectivity is the chirally symmetric parity $c\leftrightarrow q-c$, and A3 forbids
      selection among open directions before locking, so no chiral orientation is fixed by A1--A3.
      The only axiom that selects a resolved direction is A4, ``resolution occurs only when continuous
      Born--Infeld saturation meets the discrete shell support''~\cite{Beau2026m}; any chiral asymmetry
      $\pi_{LL}\neq\pi_{RR}$ must therefore arise there.
    \end{proof}

    The left-admissibility branch $\pi_{LL}=0$~\cite{Beau2026q14} is thus the maximal chiral asymmetry
    compatible with A4 locking, and the sign of $u$ is the spontaneous $V-A$ choice exchanged under
    $P_L\leftrightarrow P_R$.
    The even sector~\eqref{eq:even} is algebraic; reading it as the chirally symmetric part of a
    Born--Infeld saturation would require the identification declared unavailable above, and is not
    used here.  The parameter $u$ is the chirally asymmetric part of the locking~\cite{Beau2026c}.
    This is the structural sense in which the Higgs amplitude mode of the saturated projective condensate
    and the generation split are read from a single A4 event.

  \section{Spinorial locking and the $J_\Pi$-equivariance defect}
    \label{sec:spinorial-locking}

    This section refines Proposition~\ref{prop:strat}: the chirally symmetric branch is the transported
    equivalence $\pi_{LL}=\tau\,\overline{\pi_{RR}}\,\tau^{-1}$, not the naive block equality
    $\pi_{LL}=\pi_{RR}$, the bar being the complex conjugation carried by the antiunitary Born--Infeld
    lift $J_\Pi$.

    Let $V=\mathrm{span}(v_+,v_-)$ denote the fundamental spinorial doublet, with Cartan generator
    $Hv_\pm=\pm v_\pm$.
    The spinorial Born--Infeld lift of~\cite{Beau2026q14} is the antiunitary structure
    \begin{equation}
      \label{eq:jpidef}
      J_\Pi=\epsilon\circ\mathrm{conj},\qquad \epsilon=\sigma_z\sigma_x=i\sigma_y,
    \end{equation}
    so that, in the standard convention, $J_\Pi v_+=-v_-$ and $J_\Pi v_-=v_+$.
    Thus $J_\Pi^2=-1$ on $V$: this is the pseudoreal, or quaternionic, structure of the
    fundamental spinorial sector.

    On the symmetric square $\mathrm{Sym}^2(V)=\mathrm{span}(e_0,e_+,e_-)$ with $e_0=\sqrt{2}\,v_+v_-$,
    $e_+=v_+^2$, $e_-=v_-^2$, the induced antiunitary $J_\Pi^{(2)}$ acts by
    \begin{equation}
      \label{eq:jpisym}
      J_\Pi^{(2)}e_0=-e_0,\qquad J_\Pi^{(2)}e_+=e_-,\qquad J_\Pi^{(2)}e_-=e_+.
    \end{equation}
    Consequently $(J_\Pi^{(2)})^2=+1$ on $\mathrm{Sym}^2(V)$: the generation triplet carries a real
    involution, although the underlying doublet is pseudoreal.
    With $J_3=\mathrm{diag}(0,1,-1)$ on $(e_0,e_+,e_-)$, the formulas~\eqref{eq:jpisym} give
    \begin{equation}
      \label{eq:jpij3}
      J_\Pi^{(2)}J_3(J_\Pi^{(2)})^{-1}=-J_3,
    \end{equation}
    so the Born--Infeld spinorial conjugation exchanges the two outer generation weights.

    We now apply this to the Schur residue.
    Let $\tau:\mathcal S_R\to\mathcal S_L$ be the unitary intertwiner realising the chirality exchange
    $J_\Pi P_L=P_R J_\Pi$ of~\cite{Beau2026q14} in the chiral frame.
    Equivariance of the locking projector, $[P,J_\Pi]=0$, then forces on the diagonal chiral blocks of
    $1-P$ not the naive equality but the transported, complex-conjugated equivalence
    \begin{equation}
      \label{eq:transported}
      \pi_{LL}=\tau\,\overline{\pi_{RR}}\,\tau^{-1}.
    \end{equation}
    The relevant obstruction is consequently the antiunitary defect
    \begin{equation}
      \label{eq:defect}
      \Delta_\chi(P):=\pi_{LL}-\tau\,\overline{\pi_{RR}}\,\tau^{-1},
    \end{equation}
    which reduces to $\pi_{LL}-\tau\pi_{RR}\tau^{-1}$ on the real structure fixed by $J_\Pi^{(2)}$ on the
    generation triplet.

    \begin{lemma}[Chiral obstruction of spinorial locking]
      \label{lem:chiral-obstruction}
      Assume the projective Dirac transport is $J_\Pi$-compatible, $J_\Pi D_{g,A}J_\Pi^{-1}=\pm D_{g,A}$ and
      $J_\Pi\PiS J_\Pi^{-1}=\PiS$, as in the spinorial Born--Infeld lift of~\cite{Beau2026q14}.
      If the spinorial locking projector is $J_\Pi$-equivariant, $[P,J_\Pi]=0$, then the Schur residue
      $\EPi=-\PiS D_{g,A}(1-P)D_{g,A}\PiS^{*}$ satisfies $J_\Pi\EPi^2 J_\Pi^{-1}=\EPi^2$.
      Consequently $\langle e_+|\EPi^2|e_+\rangle=\langle e_-|\EPi^2|e_-\rangle$, and the split coefficient
      $u$ of~\eqref{eq:udef} vanishes.
      Hence a non-zero generation split requires a failure of $J_\Pi$-equivariance at projection locking,
      \begin{equation*}
        u\neq0\quad\Longrightarrow\quad [P,J_\Pi]\neq0.
      \end{equation*}
    \end{lemma}

    \begin{proof}
      Since $J_\Pi^{(2)}e_+=e_-$ and~\eqref{eq:jpij3} holds, the two outer generation weights are conjugate
      under the Born--Infeld spinorial involution.
      The assumptions $J_\Pi D_{g,A}J_\Pi^{-1}=\pm D_{g,A}$, $J_\Pi\PiS J_\Pi^{-1}=\PiS$ and $[P,J_\Pi]=0$
      imply $J_\Pi\EPi J_\Pi^{-1}=\pm\EPi$, up to the irrelevant sign induced by the chirality-odd Dirac
      transport.
      Squaring removes this sign and gives $J_\Pi\EPi^2 J_\Pi^{-1}=\EPi^2$.
      Because $J_\Pi$ is antiunitary,
      $\langle J_\Pi\psi|\EPi^2|J_\Pi\psi\rangle=\overline{\langle\psi|\EPi^2|\psi\rangle}$, and $\EPi^2$ is
      self-adjoint, so this expectation value is real.
      Taking $\psi=e_+$ and using $J_\Pi e_+=e_-$ gives
      $\langle e_-|\EPi^2|e_-\rangle=\langle e_+|\EPi^2|e_+\rangle$.
      The definition~\eqref{eq:udef} of $u$ then gives $u=0$, and the final implication is the
      contrapositive.
    \end{proof}

    \begin{remark}[Necessary condition, not a sufficient one]
      \label{rem:necessary}
      The condition $[P,J_\Pi]\neq0$ is necessary for $u\neq0$, but not sufficient.
      The commutator may be supported outside the generation triplet, or in components that do not survive
      the Dirac transport and the final projection to $\Cgen$.
      The magnitude of $u$ is therefore not determined by the abstract commutator itself: it is set by the
      $\Deff^{\pm}$-transported, generation-projected part of the antiunitary defect $\Delta_\chi(P)$
      of~\eqref{eq:defect}.
      Determining this transported and projected component is the remaining A4-level deliverable.
    \end{remark}

  \section{Locked Seeley--DeWitt conventions: operator $u$ equals spectral $u$}

    The bosonic synthesis fixes the spectral normalisation: $a_2$ is the horizontal gravitational
    response and $a_4$ the vertical gauge response of the spectral entropy functional, with induced
    couplings carrying a common factor at each Seeley--DeWitt order~\cite{Beau2026q13, Beau2026q12}.

    Three facts make the operator-level $u$ of~\eqref{eq:udef} coincide with the spectral coefficient
    that enters the effective amplitude.
    First, $u$ is quadratic in $\EPi$, so it is immune to the linear sign convention of $E$.
    Second, on the gauge-singlet $\Cgen$ the gauge curvature term vanishes, and on the flat effective
    metric $g^{\mu\nu}=2\eta^{\mu\nu}$~\cite{Beau2026q11} all curvature cross-terms vanish, in particular
    $-\tfrac16 R\EPi=0$; only $\tfrac12\EPi^2$ survives, so no linear-in-$\EPi$ term can separate the
    operator and spectral readings.
    Third, the common $a_4$ normalisation factor cancels in the generation ratio
    $(\tfrac12+u)/(\tfrac12-u)$, so $u$ is independent of the spectral normalisation while the absolute
    mass scale carries it.
    A curvature correction $-\tfrac16 R\EPi$ would reappear only on a curved background, as a
    gravitational correction to masses, and does not affect the structural $u$.

  \section{Finite locking is $J_\Pi$-equivariant: separation of the finite and Lorentzian sectors}
    \label{sec:finite-equivariance}

    Before any Lorentzian computation, the finite sector must be excluded as a source of $u$.
    This section proves that the finite projection locking is $J_\Pi$-equivariant, so that $u$ cannot
    originate in the finite cascade and the generation split is forced to be a Lorentzian datum.

    Let $\mathrm{Lock}^{\mathrm{fin}}_S$ denote the projection-locking data of axiom A4~\cite{Beau2026m} at
    finite $q$: the continuous Born--Infeld saturation condition intersected with the discrete shell
    support on $\Heis_3(\mathbb{Z}/q\mathbb{Z})$.
    Write $P_{\mathrm{fin}}=\PiS^{*}\PiS$ for the resulting finite admissible projector.

    \begin{proposition}[Finite equivariance of projection locking]
      \label{prop:finite-equivariance}
      For $q$ an odd prime, the finite locking data is invariant under the spinorial Born--Infeld parity
      $J_\Pi$, and the finite admissible configuration space carries no $J_\Pi$-odd modulus.
      Consequently
      \begin{equation}
        \label{eq:finite-equiv}
        [\mathrm{Lock}^{\mathrm{fin}}_S,J_\Pi]=0,\qquad [P_{\mathrm{fin}},J_\Pi]=0,
      \end{equation}
      and by Lemma~\ref{lem:chiral-obstruction} the finite generation split vanishes, $u_{\mathrm{fin}}=0$.
    \end{proposition}

    \begin{proof}
      Both constituents of A4 are $J_\Pi$-invariant at finite $q$.
      The continuous saturation condition is Born--Infeld parity itself, whose spinorial lift is
      $J_\Pi$~\cite{Beau2026q14}, hence $J_\Pi$-invariant by construction.
      The discrete shell support is a union of orbits of the parity
      $c\leftrightarrow q-c$~\cite{Beau2026fe}; for $q$ an odd prime this involution acts as $c\mapsto -c$
      on the central character, whose only fixed point is $c=0$, so no non-trivial shell is parity-fixed
      and the support singles out no orientation.
      The locking data is therefore $J_\Pi$-invariant.
      Equivariance of $P_{\mathrm{fin}}$ then follows from the absence of a $J_\Pi$-odd deformation: a
      cascade edge $g\to gs$ is a Heisenberg translation, not a symplectic element, so it carries no
      canonical metaplectic lift, and the chiral eigendirections are complex combinations created only by
      Lorentzian complexification~\cite{Beau2026q14}.
      There is thus no finite direction along which a $J_\Pi$-conjugate pair of locked branches could split.
      The finite locked branch is therefore $J_\Pi$-equivariant, giving~\eqref{eq:finite-equiv}.
      The vanishing $u_{\mathrm{fin}}=0$ is then Lemma~\ref{lem:chiral-obstruction}.
    \end{proof}

    The numerical front controls of Appendix~\ref{sec:controls} confirm~\eqref{eq:finite-equiv} directly,
    though they do not prove it: the proof is the absence of a finite odd modulus above.
    The symmetric front average vanishes identically, $\Theta_{\mathrm{raw}}=0$ for $q\in\{61,101,151\}$,
    and the oriented frontier is re-paired by the residual automorphism $\varphi=R_b\circ J_\Pi$, the finite
    avatar of the parity.
    These zeros are not failures to detect $u$; they are the measured signature of finite
    $J_\Pi$-equivariance.

    \begin{corollary}[Finite/Lorentzian separation]
      \label{cor:separation}

    \end{corollary}

      \mu_\chi^2>0\ \Rightarrow\ J_\Pi\text{-equivariant locking, }u=0;\qquad

  \section{Schur transversality of the A4 locking defect}
    \label{sec:schur-transversality}

    Remark~\ref{rem:necessary} left the A4-level deliverable as a necessary condition that is not yet a
    criterion: the commutator $[P,J_\Pi]\neq0$ may be supported outside the generation triplet, or may die
    under the Weyl transport and the projection to $\Cgen$.
    Section~\ref{sec:finite-equivariance} has since reduced the modulus to the Lorentzian deformation $s$
    with saturation curvature $\mu_\chi^2$~\eqref{eq:muchi}.
    We now turn the necessary condition into an exhaustive dichotomy on the transported,
    generation-projected defect.

    With $s$ the Lorentzian $J_\Pi$-odd deformation parameter of Section~\ref{sec:finite-equivariance},
    differentiating the Schur form~\eqref{eq:schur} at $s=0$ with $D_{g,A}$ fixed, and using
    $\partial_s[-(1-P)]=\partial_sP$, gives the first-order chiral polarisation
    \begin{equation}
      \label{eq:polarisation}
      \mathcal P:=\partial_s\EPi\big|_0=\PiS\,D_{g,A}\,(\partial_sP|_0)\,D_{g,A}\,\PiS^{*}.
    \end{equation}
    The positive sign is the derivative of $-(1-P)$, not a convention.
    The induced first-order contribution to $\EPi^2$ is the anticommutator $\{\EPi|_0,\mathcal P\}$ with the
    $J_\Pi$-even background; since $u$ is odd in $s$, its leading term is linear, so the vanishing
    of~\eqref{eq:polarisation} on the outer weights removes the first-order opening of the split, leaving
    the marginal branch of~\eqref{eq:muchi}.

    \begin{lemma}[Weyl-transport faithfulness on non-zero external symbols]
      \label{lem:weyl-faithful}
      Let $X$ be supported on the outer generation weights $\mathrm{span}(e_+,e_-)$.
      The transported Weyl-symbol map $X\mapsto\PiS\,D_{g,A}\,X\,D_{g,A}\,\PiS^{*}$ inherits the
      non-degeneracy of the Clifford principal symbol already used in Proposition~\ref{prop:chiralblocks}.
      It can annihilate a non-zero external symbol only when that symbol lies in the null space of the
      transported map, equivalently when its source or target component lies in the corresponding
      zero-mode sector of $\Deff^{\pm}$.
      This is the symbol-level faithfulness invoked in the proof of Proposition~\ref{prop:chiralblocks}; it
      does not assert the absence of zero modes, whose locus is classified in
      Proposition~\ref{prop:schur-dichotomy} as a Schur-null branch.
    \end{lemma}

    \begin{definition}[Schur transversality]
      \label{def:schur-transversality}
      The A4 locking defect is \emph{Schur-transverse} on the generation triplet if the chiral
      polarisation~\eqref{eq:polarisation} does not vanish on the outer weights,
      \begin{equation}
        \label{eq:transversality}
        \PiS\,D_{g,A}\,(\partial_sP|_0)\,D_{g,A}\,\PiS^{*}\big|_{\Cgen,\,J_\Pi\text{-odd}}\neq 0.
      \end{equation}
      Otherwise it is \emph{Schur-null}.
    \end{definition}

    \begin{proposition}[Schur dichotomy for the A4 locking defect]
      \label{prop:schur-dichotomy}
      Let $\sigma$ denote the projection of $[\partial_sP|_0,J_\Pi]$ onto $\mathrm{span}(e_+,e_-)$.
      On the chiral-odd A4 locus exactly one of the following alternatives holds.
      \begin{enumerate}
        \item If $\sigma=0$, the opened direction carries no $J_\Pi$-odd component on the outer generation
        weights, the polarisation~\eqref{eq:transversality} vanishes, and the locking is Schur-null.
        \item If $\sigma\neq0$ but $\sigma$ lies in the transported zero-mode locus of $\Deff^{\pm}$, the
        polarisation~\eqref{eq:transversality} again vanishes and the locking is Schur-null.
        \item If $\sigma\neq0$ and $\sigma$ does not lie in the transported zero-mode locus, the
        polarisation~\eqref{eq:transversality} is non-zero and the locking is Schur-transverse.
      \end{enumerate}
      In the Schur-null cases the first-order Schur polarisation on the outer weights is absent.
      The split does not open at first order and the saturation curvature $\mu_\chi^2$ is marginal, the
      locking falling to the marginal branch of~\eqref{eq:muchi}.
    \end{proposition}

    \begin{proof}
      By~\eqref{eq:jpisym}--\eqref{eq:jpij3} the antiunitary $J_\Pi^{(2)}$ exchanges $e_+\leftrightarrow
      e_-$, so the $J_\Pi$-odd component of $\partial_sP|_0$ on the outer weights is exactly the projection
      $\sigma$ of $[\partial_sP|_0,J_\Pi]$ onto $\mathrm{span}(e_+,e_-)$.
      If $\sigma=0$ the input to the transport~\eqref{eq:polarisation} has no surviving external-weight
      component, so the image vanishes; this implication uses only that the projection to $\Cgen$ retains
      the $e_\pm$ weights, not Weyl-transport faithfulness.
      If $\sigma\neq0$ but lies in the transported zero-mode locus, the map of
      Lemma~\ref{lem:weyl-faithful} sends it to zero.
      In both cases~\eqref{eq:transversality} vanishes, the first-order polarisation is absent, and the
      split is undecided at leading order, falling to the marginal branch.
      Conversely, if $\sigma\neq0$ and does not lie in the zero-mode locus, Lemma~\ref{lem:weyl-faithful}
      gives a non-zero transported image, so~\eqref{eq:transversality} holds.
      Faithfulness is used only in this converse direction, to prevent a non-zero external $J_\Pi$-odd
      symbol from dying under the Schur transport; the Schur-null implication does not use it.
    \end{proof}

    \begin{lemma}[Exact metaplectic opening of the Schur defect]
      \label{lem:exact-schur-opening}
      Let $V\simeq\mathbb C^2$ carry the standard $\mathfrak{sl}_2$-triple
      \[
        E=\begin{pmatrix}0&1\\0&0\end{pmatrix},\qquad
        F=\begin{pmatrix}0&0\\1&0\end{pmatrix},\qquad
        H=[E,F],
      \]
      with $H=2J_3$ on the outer generation weights of $\operatorname{Sym}^2(V)=\langle e_0,e_+,e_-\rangle$, and let
      $g(t,s)=\exp(tE)\exp(sF)$ be the oriented metaplectic step of the cascade bridge~\cite{Beau2026q14}.
      Then
      \[
        g(t,s)=\begin{pmatrix}1+ts&t\\ s&1\end{pmatrix},\qquad\det g(t,s)=1,
      \]
      and, with $\cosh r=1+\tfrac{ts}{2}$, in the non-parabolic regular domain
      \[
        \log g(t,s)=\frac{r}{\sinh r}\left(tE+sF+\frac{ts}{2}H\right).
      \]
      Consequently the $J_\Pi$-odd Cartan component transported to the outer generation doublet is
      \[
        \alpha(t,s)=ts\,\frac{r}{\sinh r}=ts+O\!\left((ts)^2\right),
      \]
      while the $R_{\mathrm{mix}}$ component vanishes identically, $\mu(t,s)\equiv0$.
      In particular $\alpha(t,s)\not\equiv0$, and $\alpha(t,s)\neq0$ for every non-degenerate oriented step
      $ts\neq0$ outside the removable logarithmic singularities.
    \end{lemma}

    \begin{proof}
      The matrix form follows from $E^2=F^2=0$, giving
      $\exp(tE)=\left(\begin{smallmatrix}1&t\\0&1\end{smallmatrix}\right)$ and
      $\exp(sF)=\left(\begin{smallmatrix}1&0\\s&1\end{smallmatrix}\right)$.
      The element $g(t,s)$ lies in $\mathrm{SL}_2$ with $\operatorname{tr}g(t,s)=2+ts$, hence
      $\cosh r=\tfrac12\operatorname{tr}g(t,s)=1+\tfrac{ts}{2}$.
      For a non-parabolic $\mathrm{SL}_2$ element the logarithm is
      $\log g=\frac{r}{\sinh r}\big(g-\tfrac12(\operatorname{tr}g)\,\mathbb 1\big)$, and here
      $g(t,s)-\tfrac12(\operatorname{tr}g)\,\mathbb 1=tE+sF+\tfrac{ts}{2}H$.
      Since $H=2J_3$, the $J_3$ coefficient is $ts\,(r/\sinh r)$, and $r/\sinh r=1+O(ts)$ gives
      $\alpha(t,s)=ts+O((ts)^2)$.
      Lifting to $\operatorname{Sym}^2(V)$, the horizontal generators $E,F$ send the outer weights $e_\pm$ to the
      central weight $e_0$, so their restriction to $\langle e_+,e_-\rangle$ carries no $R_{\mathrm{mix}}$
      component; only the Cartan $J_3$ survives, whence $\mu(t,s)\equiv0$, the mixing direction being absent in the
      real symplectic model~\cite{Beau2026q14}.
      The factor $r/\sinh r$ is strictly positive on the regular real domain, so $\alpha(t,s)=0$ only at the
      degenerate step $ts=0$.
    \end{proof}

    \begin{corollary}[Schur transversality in the present spin stratum]
      \label{cor:schur-transversality-closed}
      Assume the symbol-compatibility of the projected Dirac operator~\cite{Beau2026q14} and the Lorentzian
      identification of the central commutator $Z=[X,Y]\simeq\partial_\tau$ of the effective metric
      closure~\cite{Beau2026q11}.
      Then the A4 locking defect is Schur-transverse on the generation triplet in the present Lorentzian spin
      stratum,
      \[
        \PiS\,D_{g,A}\,(\partial_sP|_0)\,D_{g,A}\,\PiS^{*}\big|_{\Cgen,\,J_\Pi\text{-odd}}\neq0.
      \]
    \end{corollary}

    \begin{proof}
      By Lemma~\ref{lem:exact-schur-opening} the transported defect contains a non-zero $J_\Pi$-odd Cartan
      component $\sigma=\alpha(t,s)\,J_3$ for every non-degenerate oriented step, so the Schur-null case
      $\sigma=0$ is excluded.
      Its source is the commutator $Z=[X,Y]$, identified with the timelike direction $\partial_\tau$ in the
      Lorentzian effective metric, and the Clifford principal symbol is non-degenerate on non-zero timelike
      covectors.
      Therefore the transported $J_3$-symbol is not in the zero-mode locus of the Weyl blocks $\Deff^{\pm}$, so the
      transported zero-mode case is excluded.
      The remaining alternative of Proposition~\ref{prop:schur-dichotomy} is the Schur-transverse branch.
    \end{proof}

    \begin{corollary}[A4-level reduction]
      \label{cor:a4-reduction}
      The A4 residual problem is reduced to the projected Lorentzian datum
      \[
        \sigma=\big[\partial_sP|_0,J_\Pi\big]\big|_{\mathrm{span}(e_+,e_-)}.
      \]
      If the locking is Schur-transverse, the generation split is decided by the Born--Infeld genus of the
      chiral polarisation~\cite{Beau2026bim}: $\mu_\chi^2>0$ gives $u=0$, and $\mu_\chi^2<0$ gives a
      spontaneous $V-A$ split $u\neq0$.
      If the locking is Schur-null, $\mu_\chi^2=0$ and the split is governed by the leading even
      coefficient $\lambda_{2k}$ of~\eqref{eq:muchi}.
    \end{corollary}

    Schur transversality is thereby closed in the present Lorentzian spin stratum:
    Lemma~\ref{lem:exact-schur-opening} evaluates the projected A4 commutator in closed form and excludes the
    exact-zero branch, and Corollary~\ref{cor:schur-transversality-closed} excludes the transported zero-mode
    branch, so the dichotomy of Proposition~\ref{prop:schur-dichotomy} selects the Schur-transverse case.
    The Lorentzian datum $\partial_sP|_0$ is the metaplectic opening $g(t,s)=\exp(tE)\exp(sF)$, fixed together
    with the genus of~\cite{Beau2026bim} by the Born--Infeld saturation structure~\cite{Beau2026c}.
